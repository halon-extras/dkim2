CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

PROJECT("dkim2" VERSION 1.1.0)

# RPM/DEB and CENTOS8/ROCKY9/ROCKY10/UBU2204/UBU2404/UBU2604/AZURE3
IF (EXISTS "${CMAKE_CURRENT_BINARY_DIR}/OSRELEASE.txt")
	FILE(STRINGS ${CMAKE_CURRENT_BINARY_DIR}/OSRELEASE.txt OSRELEASE)
ELSEIF (EXISTS "/OSRELEASE.txt")
	FILE(STRINGS /OSRELEASE.txt OSRELEASE)
ELSE()
	MESSAGE(FATAL_ERROR "Missing OSRELEASE")
ENDIF()

IF (OSRELEASE MATCHES "CENTOS8")
	SET(CPACK_GENERATOR "RPM")
ELSEIF (OSRELEASE MATCHES "ROCKY9")
	SET(CPACK_GENERATOR "RPM")
ELSEIF (OSRELEASE MATCHES "ROCKY10")
	SET(CPACK_GENERATOR "RPM")
ELSEIF (OSRELEASE MATCHES "UBU2204")
	SET(CPACK_GENERATOR "DEB")
ELSEIF (OSRELEASE MATCHES "UBU2404")
	SET(CPACK_GENERATOR "DEB")
ELSEIF (OSRELEASE MATCHES "UBU2604")
	SET(CPACK_GENERATOR "DEB")
ELSEIF (OSRELEASE MATCHES "AZURE3")
	SET(CPACK_GENERATOR "RPM")
ELSE()
	MESSAGE(FATAL_ERROR "Invalid OSRELEASE")
ENDIF()

IF (OSRELEASE MATCHES "^(CENTOS8|ROCKY8|ROCKY9|UBU2204|UBU2404|AZURE3)$")
	SET(HALON_PACKAGE "halon")
ELSE()
	SET(HALON_PACKAGE "halon-smtpd")
ENDIF()

SET(CMAKE_BUILD_TYPE Release)

ADD_CUSTOM_TARGET(go ALL
	COMMAND ${CMAKE_COMMAND} -E env GOPATH=${GOPATH} go build -buildmode c-shared -o ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.so
	WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
)

ADD_CUSTOM_TARGET(sbom ALL
	COMMAND ${CMAKE_COMMAND} -E env GOPATH=${GOPATH} cyclonedx-gomod app -json -packages -files -licenses -output "${CMAKE_CURRENT_BINARY_DIR}/sbom.json"
	WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
)

ADD_CUSTOM_TARGET(licenses ALL
	COMMAND ${CMAKE_COMMAND} -E env GOPATH=${GOPATH} rm -rf "${CMAKE_CURRENT_BINARY_DIR}/licenses" && go-licenses save . --save_path="${CMAKE_CURRENT_BINARY_DIR}/licenses"
	WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
)

IF(EXISTS "/etc/os-release")
	EXECUTE_PROCESS(COMMAND "sed" "-ne" "s/^ID=\"\\?\\([a-z]\\+\\)\"\\?$/\\1/p" "/etc/os-release" OUTPUT_VARIABLE OS_RELEASE_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
	EXECUTE_PROCESS(COMMAND "sed" "-ne" "s/^VERSION_ID=\"\\?\\([0-9\\.]\\+\\)\"\\?$/\\1/p" "/etc/os-release" OUTPUT_VARIABLE OS_RELEASE_VERSION_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
	EXECUTE_PROCESS(COMMAND "uname" "-m" OUTPUT_VARIABLE OS_RELEASE_ARCH OUTPUT_STRIP_TRAILING_WHITESPACE)
	SET(SYSTEM_NAME "${OS_RELEASE_ID}-${OS_RELEASE_VERSION_ID}-${OS_RELEASE_ARCH}")
ENDIF()

INSTALL(FILES ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.so DESTINATION plugins)
INSTALL(FILES ${PROJECT_NAME}.functions.json DESTINATION share/plugins)
#INSTALL(FILES ${PROJECT_NAME}-app.schema.json DESTINATION share/plugins)

INSTALL(FILES "${CMAKE_CURRENT_BINARY_DIR}/sbom.json" DESTINATION share/sbom/plugins RENAME ${PROJECT_NAME}.json)

IF(${CMAKE_VERSION} VERSION_GREATER_EQUAL "3.14.0") 
	FILE(CREATE_LINK "/opt/halon/plugins/${PROJECT_NAME}.so" "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.so.link" SYMBOLIC)
ELSE()
	ADD_CUSTOM_TARGET(link_target ALL COMMAND ${CMAKE_COMMAND} -E create_symlink "/opt/halon/plugins/${PROJECT_NAME}.so" "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.so.link")
ENDIF()

INSTALL(FILES "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.so.link" DESTINATION plugins/autoload/smtpd RENAME ${PROJECT_NAME}.so)
INSTALL(FILES "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.so.link" DESTINATION plugins/autoload/hsh RENAME ${PROJECT_NAME}.so)

INSTALL(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/licenses" DESTINATION share/doc/plugins/${PROJECT_NAME})

SET(CMAKE_INSTALL_PREFIX "/opt/halon")
SET(CPACK_PACKAGING_INSTALL_PREFIX "/opt/halon")

IF (CPACK_GENERATOR MATCHES "DEB")
	SET(CPACK_DEBIAN_PACKAGE_DEPENDS "${HALON_PACKAGE} (>= 6.10)")
	SET(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/postinst;${CMAKE_CURRENT_SOURCE_DIR}/postrm")
ENDIF()
IF (CPACK_GENERATOR MATCHES "RPM")
	SET(CPACK_RPM_PACKAGE_REQUIRES "${HALON_PACKAGE} >= 6.10")
	SET(CPACK_RPM_POST_INSTALL_SCRIPT_FILE "${CMAKE_CURRENT_SOURCE_DIR}/postinst")
	SET(CPACK_RPM_POST_UNINSTALL_SCRIPT_FILE "${CMAKE_CURRENT_SOURCE_DIR}/postrm")
ENDIF()

SET(CPACK_PACKAGE_NAME "halon-extras-${PROJECT_NAME}")
SET(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
SET(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Halon extras ${PROJECT_NAME}")
SET(CPACK_PACKAGE_DESCRIPTION "Halon extras ${PROJECT_NAME}")
SET(CPACK_PACKAGE_CONTACT "Halon support@halon.io")
SET(CPACK_PACKAGE_FILE_NAME "${CPACK_PACKAGE_NAME}-${PROJECT_VERSION}-${SYSTEM_NAME}")

SET(CPACK_DEBIAN_PACKAGE_SECTION "mail")

INCLUDE(CPack)
